\section{Purpose and Scope}

\begin{legalclauses}
  \item The Parties wish to evaluate or perform the transaction, program, or
  relationship described below (the \DefinedTerm{Purpose}).

  \LSFormMultilineField[required,maxlen=1600]
    {agreement_purpose}
    {Purpose --- describe the transaction, program, or relationship}

  \item This template is a structural starting point. Counsel must tailor the
  substance to the transaction, governing law, contract requirements, data
  rights, export controls, and remedies.
\end{legalclauses}

\section{Definitions}

\begin{legalclauses}
  \item \DefinedTerm{Confidential Information} has the meaning completed below.

  \LSFormMultilineField[required,maxlen=2200]
    {agreement_confidential_information_definition}
    {Confidential Information --- define covered information, marking methods,
     oral-disclosure confirmation, and whether discussions are covered}

  \item Confidential Information does not include information the Receiving
  Party can demonstrate by contemporaneous records:
  \begin{legalclauses}
    \item was lawfully known without restriction before disclosure;
    \item becomes public without breach of this Agreement;
    \item is received lawfully from a third party without a duty of
    confidentiality; or
    \item is independently developed without use of the Disclosing Party's
    Confidential Information.
  \end{legalclauses}
\end{legalclauses}

\section{Use and Protection}

\begin{legalclauses}
  \item The Receiving Party shall use Confidential Information solely for the
  Purpose and shall disclose it only to the authorized representatives defined
  below who have a need to know and are bound by appropriate obligations.

  \LSFormMultilineField[maxlen=1200]
    {agreement_authorized_representatives}
    {Authorized representatives and required obligations}

  \item The Receiving Party shall protect Confidential Information using the
  standard of care entered below and shall promptly report the specified
  categories of unauthorized access, use, or disclosure.

  \LSFormTextField[maxlen=300]
    {agreement_standard_of_care}{Agreed standard of care}
  \LSFormMultilineField[maxlen=1200]
    {agreement_incident_reporting}{Reportable events, timing, and notice method}

  \item Nothing in this Agreement authorizes access to classified information,
  CUI, technical data subject to export controls, or other regulated
  information. Such access requires the separate governmental, contractual,
  facility, personnel, and system authorizations that apply.
\end{legalclauses}

\section{Required Disclosure}

\begin{legalclauses}
  \item If disclosure is required by law, order, or legal process, the Receiving
  Party shall follow the notice, cooperation, protective-order, and
  minimum-disclosure obligations completed below, subject to applicable law.

  \LSFormMultilineField[maxlen=1800]
    {agreement_required_disclosure_procedure}
    {Required-disclosure procedure}
\end{legalclauses}

\section{Ownership and No License}

\begin{legalclauses}
  \item Each Disclosing Party retains its rights in its Confidential
  Information. Except as expressly stated, no license or other right is granted
  by disclosure, implication, estoppel, or otherwise.

  \item Any reviewed terms addressing feedback, residuals, reverse engineering,
  export data, government data rights, or third-party materials are stated
  below.

  \LSFormMultilineField[maxlen=2000]
    {agreement_special_rights_terms}{Additional ownership, license, and data-rights terms}
\end{legalclauses}

\section{Term and Disposition}

\begin{legalclauses}
  \item This Agreement begins on the Effective Date and continues for the term
  completed below. The confidentiality and use restrictions survive as stated,
  subject to applicable law.

  \LSFormTextField[maxlen=300]{agreement_term}{Agreement term}
  \LSFormTextField[maxlen=400]
    {agreement_survival_period}{Survival period or terminating condition}

  \item On written request or termination, the Receiving Party shall handle the
  Confidential Information as completed below and, if required, certify
  completion in writing.

  \LSFormMultilineField[maxlen=1400]
    {agreement_disposition_requirements}
    {Return, destruction, archival-copy, and certification requirements}
\end{legalclauses}

\section{General Provisions}

\begin{legalclauses}
  \item \textbf{Governing Law.} This Agreement is governed by the laws entered

  venue, sovereign-immunity, or federal-law requirements.

  \LSFormTextField[required,maxlen=160]

  \LSFormMultilineField[maxlen=1200]
    {agreement_forum_venue_terms}{Forum, venue, immunity, and related terms}

  \item \textbf{Notices.} Formal notices must be delivered to the addresses and

  \LSFormTextField[maxlen=400]
    {agreement_notice_delivery_methods}{Permitted delivery methods}
  \LSFormTextField[maxlen=400]
    {agreement_notice_effectiveness}{When notice becomes effective}

  \item \textbf{Entire Agreement; Amendments.} The integration and amendment
  rules are completed below.

  \LSFormMultilineField[maxlen=1200]
    {agreement_integration_amendment_terms}{Integration rule and authorized amendment method}

  \item \textbf{Assignment; Waiver; Severability.} The reviewed provisions

  \LSFormMultilineField[maxlen=1600]
    {agreement_assignment_waiver_severability_terms}
    {Assignment, waiver, and severability terms}

  \item \textbf{Counterparts and Electronic Signatures.} This Agreement may be
  executed in counterparts and by electronic signature to the extent permitted
  by applicable law and agreed by the Parties.
\end{legalclauses}
