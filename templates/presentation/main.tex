\begin{frame}{Make the governing model explicit}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      {\fontsize{18}{23}\selectfont
      State the variables, frames, assumptions, and units before presenting a
      \LSAcronym{gnc} result. The audience should be able to tell what would
      invalidate the conclusion.\par}
      \vspace{0.24in}
      \begin{block}{Example state}
        \[
          \vect{x}=
          \begin{bmatrix}
            \delta\vect{\theta}^{\transpose} & \vect{b}^{\transpose}
          \end{bmatrix}^{\transpose}
        \]
      \end{block}
    \end{column}
    \begin{column}{0.45\textwidth}
      {\fontsize{18}{22}\selectfont
        \begin{align*}
          \vect{x}_{k+1} &= \mat{\Phi}_k\vect{x}_k + \mat{G}_k\vect{w}_k,\\
          \mat{P}_{k+1}^{-} &=
            \mat{\Phi}_k\mat{P}_{k}^{+}\mat{\Phi}_k^{\transpose}
            + \mat{G}_k\mat{Q}_k\mat{G}_k^{\transpose}.
        \end{align*}
      }
      {\fontsize{15}{19}\selectfont\color{LSDeckMint}
      One \LSAcronym{ekf} equation, one interpretation, one stated
      limitation. See \autocite{maybeck1979}.\par}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Make the hierarchy visible at a glance}
  \begin{LSBullets}
    \item Lead with the conclusion the audience must evaluate.
    \item Put evidence in descending order of decision value.
      \begin{itemize}
        \item Separate measured results from modeled projections.
        \item Put assumptions and source dates next to the claim.
      \end{itemize}
    \item End with one owner, one action, and one date.
  \end{LSBullets}
\end{frame}
